{# Shared "recommended bet" star — sits in the header bar next to the spread/O-U badge instead of the old bottom-of-card banner, so a recommended game doesn't take up extra vertical space. Its recommendation gate (kelly>0, edge>=4pts, price not too short) is intentionally stricter than the ML chip's plain edge>=3pts "differs from market" highlight in _market_chips.html — the star means "bet this," the chip tint just means "worth a look." Kept in its own macro so both call sites share the same underlying pick math without duplicating it. #} {# The probability used is the final one the bet form ends up with (game.model.final_*_prob, MLB) when available, else the raw model prob. edge_pick(game) -> abbrev of the recommended team ('' when there is no recommendation). Single source of truth for the star, its label, and the highlighted moneyline button in the details sheet. #} {% macro edge_pick(game) -%} {%- if game.model and game.status not in ('Final', 'Live') and game.odds and game.odds.home_implied is not none and game.odds.away_implied is not none -%} {%- set es_home_pct = game.odds.home_implied * 100 -%} {%- set es_favors_home = es_home_pct >= 50 -%} {%- set es_team = (game.home.abbrev or game.home.abbr) if es_favors_home else (game.away.abbrev or game.away.abbr) -%} {%- set es_team_pct = es_home_pct if es_favors_home else (100 - es_home_pct) -%} {%- set es_amer = (game.odds.home_best if es_favors_home else game.odds.away_best) | float -%} {%- if es_amer > 0 -%}{%- set es_b = es_amer / 100.0 -%}{%- elif es_amer < 0 -%}{%- set es_b = -100.0 / es_amer -%}{%- else -%}{%- set es_b = 0.0 -%}{%- endif -%} {%- set es_model_fav_prob = (game.model.final_home_prob if es_favors_home else game.model.final_away_prob) if game.model.final_home_prob is defined else (game.model.home_prob if es_favors_home else game.model.away_prob) -%} {%- set es_kelly = (es_b * es_model_fav_prob - (1 - es_model_fav_prob)) / es_b if es_b > 0 else -1 -%} {%- set es_edge = ((es_model_fav_prob * 100) - es_team_pct) | round(1) -%} {%- if es_kelly > 0 and es_edge >= 4 and es_b >= 0.25 -%}{{ es_team }}{%- endif -%} {%- endif -%} {%- endmacro %} {% macro edge_star(game) %} {% set es_pick = edge_pick(game) | trim %} {% if es_pick %} {% set es_favors_home = es_pick == (game.home.abbrev or game.home.abbr) %} {% set es_p = (game.model.final_home_prob if es_favors_home else game.model.final_away_prob) if game.model.final_home_prob is defined else (game.model.home_prob if es_favors_home else game.model.away_prob) %} {% set es_edge = ((es_p * 100) - ((game.odds.home_implied if es_favors_home else game.odds.away_implied) * 100)) | round(1) %} ★{{ es_pick }} {% endif %} {% endmacro %}
